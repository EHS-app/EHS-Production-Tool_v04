        function toggleTheme() {
            const current = document.documentElement.getAttribute('data-theme');

            document.documentElement.setAttribute('data-theme', target);

        function addRow(containerId, category, qty = 1) {
            const container = document.getElementById(containerId);
            const div = document.createElement('div');

        function addCustomItem() {

        function calculate() {
            let totalPayload = 0, totalPower = 0, totalArea = 0;
            ['rigging-list', 'fixture-list', 'led-list'].forEach(id => {
                document.querySelectorAll(`#${id} .item-row`).forEach(row => {
                    const sel = row.querySelector('select'), q = parseFloat(row.querySelector('input').value) || 0;
                    if(sel && sel.options[sel.selectedIndex]) {
                        const opt = sel.options[sel.selectedIndex];
                        totalPayload += (parseFloat(sel.value) * q);
                        totalPower += (parseFloat(opt.dataset.watt) * q);
                        totalArea += (parseFloat(opt.dataset.area) * q);
                    }
                });
            });

            const pts = parseInt(document.getElementById('point-count').value);
            const hoistSel = document.getElementById('hoist-model');
            const hoistW = parseFloat(hoistSel.value);
            const hoistWattPerUnit = parseFloat(hoistSel.options[hoistSel.selectedIndex].dataset.watt);
            const swl = parseFloat(hoistSel.options[hoistSel.selectedIndex].dataset.swl);
            const dynFactor = parseFloat(document.getElementById('dynamic-factor').value);
            
            const totalMotorPower = hoistWattPerUnit * pts;
            const totalStaticWeight = totalPayload + (pts * hoistW);
            const totalDynamicWeight = totalStaticWeight * dynFactor;
            
            document.getElementById('stat-total').innerText = totalStaticWeight.toFixed(1);
            document.getElementById('stat-power').innerText = totalPower.toLocaleString();
            document.getElementById('stat-motor-power').innerText = totalMotorPower.toLocaleString();
            document.getElementById('stat-area').innerText = totalArea.toFixed(1);
            
            const factors = distributionFactors[pts];
            const dynamicPointLoads = factors.map(f => totalDynamicWeight * f);
            const staticPointLoads = factors.map(f => totalStaticWeight * f);
            const maxVal = Math.max(...dynamicPointLoads);
            
            document.getElementById('stat-dyn-max').innerText = maxVal.toFixed(1);
            document.getElementById('stat-dyn-max').style.color = maxVal > swl ? 'var(--danger)' : 'var(--secondary)';

            document.getElementById('point-visuals').innerHTML = '';
            document.getElementById('bar-chart').innerHTML = '';
            document.getElementById('vis-row').innerHTML = '';

            factors.forEach((f, i) => {
                const sLoad = staticPointLoads[i];
                const dLoad = dynamicPointLoads[i];
                const pct = Math.round(f * 100);
                
                let statusClass = '';
                let badge = '';
                if(dLoad > swl) {
                    statusClass = 'is-danger';
                    badge = '<div class="overload-badge"><span>⚠️</span> OVERLOAD</div>';
                }

                document.getElementById('point-visuals').innerHTML += `
                    <div class="point-box ${statusClass}">
                        ${badge}
                        <span style="font-size:10px; font-weight:800; color:var(--text-muted);">POINT 0${i+1} (${pct}%)</span>
                        <span class="p-val">${dLoad.toFixed(1)}<small style="font-size:1rem; font-weight:600">kg</small></span>
                        <span class="p-dyn">Static: ${sLoad.toFixed(0)}kg</span>
                    </div>`;
                
                const sH = (sLoad / (Math.max(maxVal, swl) || 1)) * 240;
                const dH = (dLoad / (Math.max(maxVal, swl) || 1)) * 240;
                const barColor = dLoad > swl ? 'var(--danger)' : 'var(--secondary)';

                document.getElementById('bar-chart').innerHTML += `
                    <div class="bar-group">
                        <div class="bars-side-by-side">
                            <div class="bar" style="height:${sH}px; background:var(--static-bar)" data-value="${sLoad.toFixed(0)}"></div>
                            <div class="bar" style="height:${dH}px; background:${barColor}" data-value="${dLoad.toFixed(0)}"></div>
                        </div>
                    </div>`;

                document.getElementById('vis-row').innerHTML += `
                    <div class="vis-point">
                        <div class="vis-arrow"></div>
                        <span class="vis-pct">${pct}%</span>
                        <span class="vis-label">P${i+1}</span>
                    </div>`;
            });
        }
        window.onload = () => { addRow('rigging-list', 'Truss', 4); };
